2. Rewrite src/pages/UploadPage.tsx to integrate all components:
   ```typescript
   import { useState } from 'react';
   import { Box, Typography, Button } from '@mui/material';
   import CloudUploadIcon from '@mui/icons-material/CloudUpload';
   import SearchIcon from '@mui/icons-material/Search';
   import { ArtistInfo } from '../types/extraction';
   import { PosterUpload } from '../components/PosterUpload';
   import { FestivalSearch } from '../components/FestivalSearch';
   import { EditableArtistList } from '../components/EditableArtistList';
   import './UploadPage.css';

   type InputMode = 'upload' | 'search';

   interface ExtractionResult {
     artists: ArtistInfo[];
     warning?: string;
     festivalName?: string;
     year?: number;
     sources?: string[];
   }

   export function UploadPage() {
     const [mode, setMode] = useState<InputMode>('upload');
     const [result, setResult] = useState<ExtractionResult | null>(null);
     const [editedArtists, setEditedArtists] = useState<ArtistInfo[]>([]);

     const handleExtractionComplete = (artists: ArtistInfo[], warning?: string) => {
       setResult({ artists, warning });
       setEditedArtists(artists);
     };

     const handleSearchComplete = (
       artists: ArtistInfo[],
       festivalName: string,
       year: number,
       sources: string[]
     ) => {
       setResult({ artists, festivalName, year, sources });
       setEditedArtists(artists);
     };

     const handleArtistsChange = (artists: ArtistInfo[]) => {
       setEditedArtists(artists);
     };

     const handleContinue = () => {
       // Store artists for next phase (Phase 4: Track Selection)
       // For now, log to console - will be connected to routing/state in Phase 4
       console.log('Proceeding with artists:', editedArtists);
       alert(`Ready to create playlist with ${editedArtists.length} artists!\n\n(Track selection coming in Phase 4)`);
     };

     const handleReset = () => {
       setResult(null);
       setEditedArtists([]);
     };

     return (
       <div className="upload-page">
         <div className="upload-container">
           <h1 className="upload-title">Create Your Festival Playlist</h1>
           <p className="upload-subtitle">
             {!result
               ? 'Upload a poster or search for a festival'
               : `${editedArtists.length} artist${editedArtists.length !== 1 ? 's' : ''} ready`
             }
           </p>

           {!result && (
             <>
               <Box className="mode-tabs">
                 <button
                   className={`mode-tab ${mode === 'upload' ? 'active' : ''}`}
                   onClick={() => setMode('upload')}
                 >
                   <CloudUploadIcon fontSize="small" />
                   Upload Poster
                 </button>
                 <button
                   className={`mode-tab ${mode === 'search' ? 'active' : ''}`}
                   onClick={() => setMode('search')}
                 >
                   <SearchIcon fontSize="small" />
                   Search Festival
                 </button>
               </Box>

               <Box className="input-section">
                 {mode === 'upload' ? (
                   <PosterUpload
                     onExtractionComplete={handleExtractionComplete}
                     onSwitchToSearch={() => setMode('search')}
                   />
                 ) : (
                   <FestivalSearch
                     onSearchComplete={handleSearchComplete}
                     onSwitchToUpload={() => setMode('upload')}
                   />
                 )}
               </Box>
             </>
           )}

           {result && (
             <Box className="results-section">
               <Box className="results-header">
                 <Typography className="results-title">
                   {result.festivalName
                     ? `${result.festivalName} ${result.year}`
                     : 'Extracted Artists'
                   }
                 </Typography>
                 <Button size="small" onClick={handleReset}>
                   Start Over
                 </Button>
               </Box>

               {result.warning && (
                 <Box className="warning-message">
                   {result.warning} You can add missing artists manually below.
                 </Box>
               )}

               <EditableArtistList
                 initialArtists={result.artists}
                 onChange={handleArtistsChange}
               />

               {result.sources && result.sources.length > 0 && (
                 <Box className="sources-section">
                   <Typography className="sources-title">Sources</Typography>
                   <ul className="sources-list">
                     {result.sources.slice(0, 3).map((source, i) => (
                       <li key={i}>
                         <a href={source} target="_blank" rel="noopener noreferrer">
                           {new URL(source).hostname}
                         </a>
                       </li>
                     ))}
                   </ul>
                 </Box>
               )}

               <Box className="continue-section">
                 <Button
                   variant="contained"
                   size="large"
                   className="continue-button"
                   disabled={editedArtists.length === 0}
                   onClick={handleContinue}
                   sx={{
                     backgroundColor: '#1DB954',
                     '&:hover': { backgroundColor: '#1ed760' }
                   }}
                 >
                   Continue with {editedArtists.length} Artist{editedArtists.length !== 1 ? 's' : ''}
                 </Button>
               </Box>
             </Box>
           )}
         </div>
       </div>
     );
   }
   ```
