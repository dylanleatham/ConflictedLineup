1. Create src/components/FestivalSearch.tsx:
   ```typescript
   import { useState } from 'react';
   import {
     Box,
     TextField,
     Button,
     Typography,
     CircularProgress,
     Alert,
     InputAdornment
   } from '@mui/material';
   import SearchIcon from '@mui/icons-material/Search';
   import { ArtistInfo } from '../types/extraction';
   import { searchFestivalLineup } from '../services/extractionApi';
   import './FestivalSearch.css';

   interface FestivalSearchProps {
     onSearchComplete: (artists: ArtistInfo[], festivalName: string, year: number, sources: string[]) => void;
     onSwitchToUpload: () => void;
   }

   export function FestivalSearch({ onSearchComplete, onSwitchToUpload }: FestivalSearchProps) {
     const [query, setQuery] = useState('');
     const [year, setYear] = useState<string>(new Date().getFullYear().toString());
     const [loading, setLoading] = useState(false);
     const [error, setError] = useState<string | null>(null);

     const handleSearch = async () => {
       const trimmedQuery = query.trim();
       if (!trimmedQuery) {
         setError('Please enter a festival name');
         return;
       }

       setError(null);
       setLoading(true);

       try {
         const yearNum = parseInt(year, 10);
         const result = await searchFestivalLineup(
           trimmedQuery,
           !isNaN(yearNum) ? yearNum : undefined
         );

         if (result.artists.length === 0) {
           setError(`No lineup found for "${trimmedQuery}". Try uploading a poster instead.`);
         } else {
           onSearchComplete(
             result.artists,
             result.festivalName,
             result.year,
             result.sources
           );
         }
       } catch (err) {
         setError(err instanceof Error ? err.message : 'Search failed. Please try again.');
       } finally {
         setLoading(false);
       }
     };

     const handleKeyPress = (e: React.KeyboardEvent) => {
       if (e.key === 'Enter' && !loading) {
         handleSearch();
       }
     };

     const currentYear = new Date().getFullYear();
     const yearOptions = Array.from({ length: 3 }, (_, i) => currentYear + i);

     return (
       <Box className="festival-search">
         <Box className="search-form">
           <TextField
             fullWidth
             label="Festival Name"
             placeholder="e.g., Coachella, Lollapalooza, Glastonbury"
             value={query}
             onChange={(e) => setQuery(e.target.value)}
             onKeyPress={handleKeyPress}
             disabled={loading}
             InputProps={{
               startAdornment: (
                 <InputAdornment position="start">
                   <SearchIcon sx={{ color: 'rgba(255,255,255,0.5)' }} />
                 </InputAdornment>
               )
             }}
           />

           <TextField
             select
             label="Year"
             value={year}
             onChange={(e) => setYear(e.target.value)}
             disabled={loading}
             SelectProps={{ native: true }}
             className="year-select"
           >
             {yearOptions.map(y => (
               <option key={y} value={y}>{y}</option>
             ))}
           </TextField>

           <Button
             variant="contained"
             onClick={handleSearch}
             disabled={loading || !query.trim()}
             className="search-button"
           >
             {loading ? (
               <>
                 <CircularProgress size={20} sx={{ mr: 1, color: 'inherit' }} />
                 Searching...
               </>
             ) : (
               'Search'
             )}
           </Button>
         </Box>

         {loading && (
           <Box className="search-status">
             <Typography variant="body2" color="textSecondary">
               Searching the web for "{query}" {year} lineup...
             </Typography>
             <Typography variant="caption" color="textSecondary" sx={{ mt: 1, display: 'block' }}>
               This may take up to 30 seconds
             </Typography>
           </Box>
         )}

         {error && (
           <Alert
             severity="warning"
             sx={{ mt: 2 }}
             action={
               error.includes('No lineup found') ? (
                 <Button color="inherit" size="small" onClick={onSwitchToUpload}>
                   Upload Poster
                 </Button>
               ) : undefined
             }
           >
             {error}
           </Alert>
         )}
       </Box>
     );
   }
   ```
