2. Create services/trackSelectionApi.ts:

```typescript
const API_URL = import.meta.env.VITE_API_URL || 'http://localhost:5000';

export async function selectTracksForArtists(
  artistNames: string[],
  spotifyAccessToken: string
): Promise<TrackSelectionResponse> {
  const response = await fetch(`${API_URL}/api/tracks/select`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      artistNames,
      spotifyAccessToken
    })
  });

  if (!response.ok) {
    const error = await response.json();
    throw new Error(error.error || 'Track selection failed');
  }

  return response.json();
}
```
